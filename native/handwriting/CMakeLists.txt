cmake_minimum_required(VERSION 3.18)
project(InkPickHandwriting LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(inkpick-handwriting helper.cpp vendor/zinnia/character.cpp
  vendor/zinnia/feature.cpp vendor/zinnia/recognizer.cpp vendor/zinnia/sexp.cpp)
target_include_directories(inkpick-handwriting PRIVATE vendor/zinnia)
target_compile_definitions(inkpick-handwriting PRIVATE WORDS_LITENDIAN ZINNIA_STATIC_LIBRARY)
if(MSVC)
  set_property(TARGET inkpick-handwriting PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_definitions(inkpick-handwriting PRIVATE _CRT_SECURE_NO_WARNINGS HAVE_WINDOWS_H HAVE_SYS_TYPES_H HAVE_SYS_STAT_H HAVE_FCNTL_H HAVE_STRING_H)
  target_link_libraries(inkpick-handwriting PRIVATE bcrypt)
else()
  find_package(OpenSSL REQUIRED)
  target_compile_definitions(inkpick-handwriting PRIVATE HAVE_SYS_TYPES_H HAVE_SYS_STAT_H HAVE_SYS_MMAN_H HAVE_FCNTL_H HAVE_UNISTD_H HAVE_MMAP)
  target_link_libraries(inkpick-handwriting PRIVATE OpenSSL::Crypto)
endif()
